# Debian packages in the Mimi USB image, installed by scripts/image/build.sh with
# mmdebstrap (variant "apt": Debian's essential packages plus apt) and without
# Recommends, so this list is the whole system. One package per line; # starts a comment.
# Components: main and non-free-firmware (see docs/LICENSING.md#bootable-usb-image).

# --- Base system
systemd-sysv
systemd-repart                # grows the root partition to fill the stick on first boot
systemd-timesyncd             # sets the clock once online (downloads use TLS)
udev
dbus
dbus-user-session             # the kiosk user's session bus (audio, browser)
libpam-systemd                # logind sessions: gives the kiosk the screen, input, sound and camera
kmod
iproute2
procps
e2fsprogs
dosfstools
ca-certificates
curl
tzdata
zstd                          # initramfs compression

# --- Kernel, CPU microcode, boot loaders (UEFI with Secure Boot, and legacy BIOS)
linux-image-amd64
initramfs-tools
grub2-common
grub-pc-bin
grub-efi-amd64-bin
grub-efi-amd64-signed
shim-signed
intel-microcode
amd64-microcode

# --- Firmware: the set Debian's live images ship for PCs (live-task-non-free-firmware-pc),
#     plus the graphics and Wi-Fi firmware firmware-misc-nonfree recommends
firmware-linux-free
firmware-misc-nonfree
firmware-amd-graphics
firmware-intel-graphics
firmware-intel-misc
firmware-nvidia-graphics
firmware-iwlwifi
firmware-realtek
firmware-atheros
firmware-brcm80211
firmware-mediatek
firmware-intel-sound
firmware-sof-signed
wireless-regdb

# --- Network (wired, Wi-Fi, phone USB tethering)
network-manager
wpasupplicant

# --- Kiosk: a Wayland compositor that runs one app full screen, and the browser
cage
chromium
fonts-noto-core
fonts-noto-color-emoji

# --- Graphics: Mesa (OpenGL for the compositor and browser, Vulkan for llama.cpp on AMD/Intel)
libgl1-mesa-dri
libegl-mesa0
libgles2
libvulkan1
mesa-vulkan-drivers

# --- Sound for voice chat (microphone and speech) in the browser
pipewire
pipewire-pulse
wireplumber
alsa-ucm-conf

# --- Mimi: the CPU llama.cpp build needs OpenMP; aria2 speeds up large downloads;
#     the installer unpacks a pinned Node.js (.tar.xz) to build the app
libgomp1
aria2
xz-utils
